go athena

# ==============================================================================
# DEFINIÇÃO DA MALHA
# ==============================================================================

# ------------------------------------------------------------------------------
# Malha no Eixo X (Horizontal)
# ------------------------------------------------------------------------------
line x loc=0.00 spac=0.10
# Início da Fonte / Borda externa da janela ativa

line x loc=1.0 spac=0.02
# Interface Fonte/Canal

line x loc=3.5 spac=0.01
# Meio do Canal

# ------------------------------------------------------------------------------
# Malha no Eixo Y (Vertical - Profundidade)
# ------------------------------------------------------------------------------
line y loc=0.000 spac=0.01
# Interface SiO2/Si e Camada de Inversão

line y loc=0.030 spac=0.02
# Pico da dopagem de ajustamento de Vt

line y loc=0.100 spac=0.05
# Região da Junção PN de Fonte/Dreno

line y loc=0.300 spac=0.20
# Transição suave para o substrato

line y loc=1.000 spac=0.150
# Fundo do Substrato / Bulk (Malha grossa para economizar nós)


# ==============================================================================
# INICIALIZAÇÃO DO SUBSTRATO
# Lâmina: Silício tipo A (P), com dopagem B (8e15 cm^-3)
# ==============================================================================
init silicon c.boron=8.0e15 orientation=100


# ==============================================================================
# ETAPA 1: DEPOSIÇÃO DO ÓXIDO DE CAMPO
# Enunciado: Deposição de 500 nm de óxido de silício.
# ==============================================================================
deposit oxide thick=0.50 spaces=5


# ==============================================================================
# ETAPA 2: FOTOGRAVAÇÃO 1
# Enunciado: Fotogravação 1 – abertura da região ativa.
# ==============================================================================
etch oxide right p1.x=1.0


# ==============================================================================
# ETAPA 3: OXIDAÇÃO TÉRMICA DE PORTA
# Enunciado: Oxidação térmica de porta de 20 nm em 900 ºC e 40 minutos.
# ==============================================================================
diffus time=65 minutes temp=900 dryo2


# ==============================================================================
# ETAPA 4: AJUSTE DE TENSAO DE LIMIAR (VT)
# Enunciado: Implantação Iônica para ajuste de tensão de limiar com energia de 30 keV.
# ==============================================================================
implant phosphor dose=1.0e11 energy=30 tilt=0 rotation=0 crystal

# Salvamento e exibição da estrutura do Passo 4
structure outfile=pmos_Passo4.str
tonyplot pmos_Passo4.str


# ==============================================================================
# ETAPA 5: PASSO OPCIONAL (LDD)
# Enunciado: Etapa a ser adicionada para as questões Q2 e Q3.
# ==============================================================================


# ==============================================================================
# ETAPA 6: DEPOSIÇÃO DO SILÍCIO POLICRISTALINO
# Enunciado: Deposição de Silício policristalino intrínseco (400 nm) em 600 ºC e 15 minutos.
# ==============================================================================
deposit polysilicon thick=0.4


# ==============================================================================
# ETAPA 7: DOPAGEM DO POLISILÍCIO DE PORTA
# Enunciado: Difusão de C (BOR = Boro) para dopagem degenerada do silício policristalino de porta.
# ==============================================================================
diffus time=15 minutes temp=900 nitro c.boron=1e20


# ==============================================================================
# ETAPA 8: FOTOGRAVAÇÃO 2
# Enunciado: Fotogravação 2 – definição do silício policristalino.
# ==============================================================================
etch polysilicon left p1.x=1
etch oxide left p1.x=1.0

etch start x=0.9 y=-1
etch continue x=0.9 y=-0.412
etch continue x=1.6 y=-0.412
etch end x=1.6 y=-1.6


# ==============================================================================
# ETAPA 9: IMPLANTAÇÃO DE FONTE E DRENO
# Enunciado: Implantação de D (FOS = Fósforo) de Fonte/Dreno com energia E (60 keV) e dose F (4e15 cm^-2).
# ==============================================================================
implant phosphor dose=4.0e15 energy=60 tilt=0 rotation=0 crystal

# Salvamento e exibição da estrutura do Passo 9
structure outfile=pmos_Passo9.str
tonyplot pmos_Passo9.str


# ==============================================================================
# ETAPA 10: RECOZIMENTO DE ATIVAÇÃO
# Enunciado: Recozimento em forno com ambiente de N2 em 940 ºC e 20 minutos para ativação das impurezas implantadas.
# ==============================================================================
diffus time=20 minutes temp=940 nitro


# ==============================================================================
# ETAPA 11: DEPOSIÇÃO DE ÓXIDO DE PASSIVAÇÃO
# Enunciado: Deposição de 400 nm de óxido de silício.
# ==============================================================================
deposit oxide thick=0.4


# ==============================================================================
# ETAPA 12: DENSIFICAÇÃO DO ÓXIDO
# Enunciado: Densificação do óxido em forno com ambiente de N2 em 940 ºC e 10 minutos.
# ==============================================================================
diffus time=10 temp=940 nitro


# ==============================================================================
# ETAPA 13: FOTOGRAVAÇÃO 3
# Enunciado: Fotogravação 3 – abertura de contatos.
# ==============================================================================
etch start x=0.5 y=-1.4
etch continue x=0.5 y=-0.4
etch continue x=3.5 y=-0.4
etch end x=3.5 y=-1.3

etch start x=0.25 y=-0.4
etch continue x=0.25 y=0
etch continue x=0.75 y=0
etch end x=0.75 y=-0.4

# Salvamento e exibição da estrutura do Passo 13
structure outfile=pmos_Passo13.str
tonyplot pmos_Passo13.str


# ==============================================================================
# ETAPA 14: DEPOSIÇÃO DE TITÂNIO
# Enunciado: Deposição de 100 nm de Titânio (barreira de difusão).
# ==============================================================================
deposit Titanium thick=0.1

# Definição e corrosão da camada de Titânio
etch start x=0 y=-1.6
etch continue x=0 y=-0.4
etch continue x=1 y=-0.4
etch end x=1 y=-1.6

etch start x=0.25 y=-0.4
etch continue x=0.25 y=-0.1
etch continue x=0.35 y=-0.1
etch end x=0.35 y=-0.4

etch start x=0.65 y=-0.4
etch continue x=0.65 y=-0.1
etch continue x=0.75 y=-0.1
etch end x=0.75 y=-0.4


# ==============================================================================
# ETAPA 15: DEPOSIÇÃO DE ALUMÍNIO
# Enunciado: Deposição de 1000 nm de Alumínio.
# ==============================================================================
deposit Aluminum thick=1

# Definição e corrosão da camada de Alumínio
etch start x=0 y=-1.6
etch continue x=0 y=-0.4
etch continue x=0.25 y=-0.4
etch end x=0.25 y=-1.6

etch start x=0.75 y=-1.6
etch continue x=0.75 y=-0.4
etch continue x=1 y=-0.4
etch end x=1  y=-1.6

etch start x=0 y=-1.6
etch continue x=0 y=-1.1
etch continue x=1 y=-1.1
etch end x=1 y=-1.6


# ==============================================================================
# ETAPA 16: FOTOGRAVAÇÃO 4
# Enunciado: Fotogravação 4 – definição do Alumínio e Titânio.
# ==============================================================================


# ==============================================================================
# ETAPA 17: SINTERIZAÇÃO
# Enunciado: Sinterização em 420 ºC e 30 minutos.
# ==============================================================================

# Salvamento e exibição da estrutura final do Passo 17
structure outfile=pmos_Passo17.str
tonyplot pmos_Passo17.str

# ==============================================================================
# ETAPA FINAL: ESPELHAMENTO DA ESTRUTURA
# ==============================================================================
# Espelha a malha em torno da posição x especificada (ex: eixo central da porta)
struct mirror right


#salva a estrutura completa para o Atlas
structure outfile=pmos_final_completo.str
tonyplot pmos_final_completo.str
